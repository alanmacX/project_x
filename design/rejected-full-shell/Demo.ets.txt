  private updateBars():void {if(!this.context)return;const dark=this.screen!=='home';window.getLastWindow(this.context).then(w=>w.setWindowSystemBarProperties({statusBarContentColor:dark?'#FFFFFF':'#24251F',navigationBarContentColor:dark?'#FFFFFF':'#24251F'})).catch(()=>{});}

    try{this.context=this.getUIContext().getHostContext() as common.UIAbilityContext;
      await this.store.init(this.context);await this.library.init(this.context);this.assets=this.library.items.slice();
      for(const definition of CARD_TEMPLATES){const c=makeTemplateCards(definition.id,1,()=>definition.id)[0];if(c){if(definition.id==='portrait'){this.attachPhoto(c,this.assets[0],false);await this.suggestClock(c);}this.previews.set(definition.id,c);}}
      await this.refreshWorks();this.recoverable=fileIo.accessSync(this.draftPath());this.ready=true;this.updateBars();

  private draftPath():string{return this.context!.filesDir+'/demo-draft.json';}
  private readDrafts():DemoDraft[]{if(!fileIo.accessSync(this.draftPath()))return [];const raw=JSON.parse(fileIo.readTextSync(this.draftPath())) as DemoDraftFile;return Array.isArray(raw.items)?raw.items:raw.id&&raw.state?[{id:raw.id,state:raw.state}]:[];}
  private flushDraft():void{
    if(this.draftTimer>=0)clearTimeout(this.draftTimer);this.draftTimer=-1;
    if(!this.context||!['board','edit'].includes(this.screen))return;
    const draft=new DemoDraft();draft.id=this.document.canvasId;draft.state=JSON.parse(JSON.stringify(this.document)) as FridgeState;
    // Nested customization is recoverable as part of its parent work.
    if(this.screen==='edit'){const i=draft.state.cards.findIndex(c=>c.id===this.face.id);if(i>=0)draft.state.cards[i]=cardViewSnapshot(this.face);}
    try{const data=new DemoDraftFile();data.items=this.readDrafts().filter(d=>d.id!==draft.id).concat([draft]);writeDemoFile(this.draftPath(),JSON.stringify(data));}catch(e){this.toast('草稿保存失败：'+(e as Error).message);}
  }
  private queueDraft():void{if(this.draftTimer>=0)clearTimeout(this.draftTimer);this.draftTimer=setTimeout(()=>this.flushDraft(),180);}
  private clearDraft():void{if(this.draftTimer>=0)clearTimeout(this.draftTimer);this.draftTimer=-1;const data=new DemoDraftFile();data.items=this.readDrafts().filter(d=>d.id!==this.document.canvasId);if(data.items.length)writeDemoFile(this.draftPath(),JSON.stringify(data));else if(fileIo.accessSync(this.draftPath()))fileIo.unlinkSync(this.draftPath());this.recoverable=data.items.length>0;}
  private async recover():Promise<void>{
    try{const draft=this.readDrafts()[0];if(!draft)return;await this.openWork(draft.id);this.document=draft.state;this.selected='';this.screen='board';this.resetHistory();this.recoverable=false;}catch(e){this.toast((e as Error).message);}
  }
  private resetHistory():void{this.undoStack=[];this.redoStack=[];this.undoCount=0;this.redoCount=0;}
  private checkpoint():void{this.undoStack.push(JSON.stringify(this.document));if(this.undoStack.length>40)this.undoStack.shift();this.redoStack=[];this.undoCount=this.undoStack.length;this.redoCount=0;}
  private undo(redo:boolean=false):void{
    const from=redo?this.redoStack:this.undoStack,to=redo?this.undoStack:this.redoStack;if(!from.length)return;
    to.push(JSON.stringify(this.document));this.document=JSON.parse(from.pop()!) as FridgeState;this.undoCount=this.undoStack.length;this.redoCount=this.redoStack.length;if(!this.document.cards.some(c=>c.id===this.selected))this.selected='';this.queueDraft();
  }
  private async openWork(id:string):Promise<void>{
    if(this.busy)return;this.busy=true;
    try{await this.store.selectCanvas(id,false);this.document=JSON.parse(storageSnapshot(this.store.state)) as FridgeState;this.workName=this.store.activeName;this.selected='';this.screen='board';this.resetHistory();}
    catch(e){this.toast((e as Error).message);}finally{this.busy=false;}
  }
  private async createWork(templateId:string=''):Promise<void>{
    if(this.busy)return;this.busy=true;
    try{this.store.createCanvas(templateDefinition(templateId)?.name??'自由创作',undefined,this.newDimension);this.store.state.cards=[];this.store.state.background.color='#EEE9DF';
      if(templateId){const card=makeTemplateCards(templateId,this.store.state.canvasAspect,()=>this.store.newId())[0];card.x=0;card.y=0;card.w=344;card.h=344/this.store.state.canvasAspect;card.template!.fill=true;
        if(templateId==='portrait'){this.attachPhoto(card,this.assets[0],false);await this.suggestClock(card);}this.store.addCard(card);this.store.state.background.color=templateColors(card)[0];}
      this.store.state=await projectCapabilityData(this.store.state,this.context!);this.store.adoptCanvas(this.store.state);await this.store.save(false);this.document=JSON.parse(storageSnapshot(this.store.state)) as FridgeState;this.workName=this.store.activeName;this.screen='board';this.resetHistory();
      if(templateId)this.customize(this.document.cards[0].id);this.sheet=false;
    }catch(e){this.toast((e as Error).message);}finally{this.busy=false;}
  }
  private async saveWork(desktop:boolean=false):Promise<void>{
    if(this.busy)return;this.busy=true;
    try{this.store.renameCanvas(this.workName,this.document.canvasId);this.store.state=this.document;this.store.adoptCanvas(this.document);await this.store.save(true);this.clearDraft();await this.refreshWorks();this.screen=desktop?'desktop':'home';this.tab=0;this.desktopStatus='';}
    catch(e){this.toast('保存失败：'+(e as Error).message);}finally{this.busy=false;}
  }
  private publishDocument():void{this.document=JSON.parse(JSON.stringify(this.document)) as FridgeState;this.queueDraft();}

  private customize(id:string):void{this.tick=Date.now();const card=this.document.cards.find(c=>c.id===id);if(!card)return;this.selected=id;this.face=cardViewSnapshot(card);this.nestedBefore=JSON.stringify(card);this.editTab=0;const photo=card.elements.find(e=>e.kind==='image'),asset=this.library.match(photo?.src??'');this.toneIndex=Math.max(0,asset?.tones.indexOf(photo?.src??'')??0);this.screen='edit';this.camera.reset();}
  private publishFace():void{this.face=cardViewSnapshot(this.face);this.queueDraft();}
  private finishFace():void{if(JSON.stringify(this.face)!==this.nestedBefore){this.checkpoint();const i=this.document.cards.findIndex(c=>c.id===this.face.id);if(i>=0)this.document.cards[i]=cardViewSnapshot(this.face);this.publishDocument();}this.screen='board';this.queueDraft();}

  private currentPhoto():CanvasElement|undefined{return this.face.elements.find(e=>e.kind==='image');}

  private attachPhoto(card:FridgeCard,item:DemoAsset,subject:boolean):void{
    const element=card.elements.find(e=>e.kind==='image')??new CanvasElement();if(!element.id){element.id=this.store.newId();card.elements.push(element);}element.kind='image';element.src=subject&&item.subject?item.subject:item.original;element.x=0;element.y=0;element.w=1;element.h=1;element.depthSourceAspect=subject?item.aspect*item.crop.w/item.crop.h:item.aspect;
    element.depthEnabled=!subject&&!!item.subject;element.depthForeground=subject?'':item.subject;element.depthCrop=JSON.parse(JSON.stringify(item.crop)) as ElementBox;
    if(card.template){card.template.photoMode='photo';card.template.photoFraming='cover';card.template.zoom=1;card.template.offsetX=0;card.template.offsetY=0;card.template.depth=!!item.subject;}
  }
  private async suggestClock(card:FridgeCard):Promise<void>{
    const photo=card.elements.find(e=>e.kind==='image'),t=card.template;if(!photo||!t||t.id!=='portrait')return;
    try{const subject=photo.depthForeground?await readPhotoSamples(photo.depthForeground):new PhotoSamples(),original=await readPhotoSamples(photo.src),choice=recommendPhotoClock(card,subject,original,templateColors(card)[0]);
      t.clockPosition=choice.position;t.clockInk=choice.ink;t.clockAuto=true;
      const frame=portraitPhotoFrame(card,photo),box=photoClockLayout(300,300*card.h/card.w,choice.position),crop=photo.depthCrop;let hits=0,total=0;
      for(let y=0;y<10;y++)for(let x=0;x<12;x++){const px=box.x+(x+.5)*box.w/12,py=box.y+(y+.5)*box.h/10,u=((px-frame.x)/frame.w-crop.x)/crop.w,v=((py-frame.y)/frame.h-crop.y)/crop.h;if(subject.width&&u>=0&&u<1&&v>=0&&v<1)hits+=(subject.rgba[(Math.floor(v*subject.height)*subject.width+Math.floor(u*subject.width))*4+3]??0)/255;total++;}
      t.depth=!!photo.depthForeground&&hits/total<.35;
    }catch(_){t.clockPosition='bottom-left';t.clockInk='#FFFFFF';t.depth=false;}
  }
  private async smartClock():Promise<void>{const id=this.face.id,before=JSON.stringify(this.face),next=cardViewSnapshot(this.face);await this.suggestClock(next);if(this.screen==='edit'&&this.face.id===id&&JSON.stringify(this.face)===before){this.face=next;this.publishFace();}}
  private addTemplate(id:string):void{if(this.document.cards.length>=16){this.toast('每件作品最多 16 张贴纸');return;}this.checkpoint();const card=makeTemplateCards(id,this.document.canvasAspect,()=>this.store.newId())[0];
    const scale=Math.min(344*.65/card.w,344/this.document.canvasAspect*.65/card.h);card.w*=scale;card.h*=scale;card.x=(344-card.w)/2;card.y=(344/this.document.canvasAspect-card.h)/2/(344/this.document.canvasAspect)*470;card.z=++this.document.maxZ;
    if(id==='portrait'){this.attachPhoto(card,this.assets[0],false);const preset=this.previews.get(id)?.template;if(preset){card.template!.clockPosition=preset.clockPosition;card.template!.clockInk=preset.clockInk;card.template!.depth=preset.depth;}}this.document.cards.push(card);this.selected=card.id;this.sheet=false;this.publishDocument();this.customize(card.id);
  }
  private addText():void{if(this.document.cards.length>=16){this.toast('每件作品最多 16 张贴纸');return;}this.checkpoint();const card=new FridgeCard();card.id=this.store.newId();card.x=72;card.y=180;card.w=200;card.h=140;card.material='cream';card.z=++this.document.maxZ;const e=new CanvasElement();e.id=this.store.newId();e.text='把喜欢的留在这里';e.fs=24;e.x=.08;e.y=.18;e.w=.84;e.h=.64;e.align='center';card.elements=[e];this.document.cards.push(card);this.publishDocument();this.customize(card.id);this.editTab=2;this.sheetType='text';}
  private openMaterials(target:string):void{this.materialTarget=target;this.materialOwner=this.screen==='edit'?this.face.id:this.document.canvasId;this.subjectMaterial=false;this.sheetType='materials';this.sheet=true;}
  private useMaterial(item:DemoAsset):void{
    if(this.materialTarget==='library'){this.sheetType='asset';this.contentIndex=this.assets.findIndex(a=>a.id===item.id);return;}
    if(this.materialTarget==='album'&&this.face.id===this.materialOwner&&this.face.capability?.k==='album'){this.face.capability.albumCover=item.original;this.face.capability.albumBackground=item.original;this.publishFace();this.sheet=false;return;}
    if(this.materialTarget==='face'&&this.face.id===this.materialOwner){this.attachPhoto(this.face,item,this.subjectMaterial);this.publishFace();this.smartClock();this.sheet=false;return;}
    if(this.materialTarget==='board'&&this.document.canvasId===this.materialOwner){
      if(this.document.cards.length>=16){this.toast('每件作品最多 16 张贴纸');return;}this.checkpoint();const card=new FridgeCard();card.id=this.store.newId();card.w=210;card.h=Math.min(250,210/item.aspect);card.x=(344-card.w)/2;card.y=(344/this.document.canvasAspect-card.h)/2/(344/this.document.canvasAspect)*470;card.z=++this.document.maxZ;this.attachPhoto(card,item,this.subjectMaterial);this.document.cards.push(card);this.selected=card.id;this.publishDocument();this.sheet=false;
    }
  }
  private async importPhoto():Promise<void>{if(this.busy)return;this.busy=true;const target=this.materialTarget,owner=this.materialOwner;
    try{const item=await this.library.importPhoto();this.assets=this.library.items.slice();if(item&&target===this.materialTarget&&owner===this.materialOwner&&target!=='library')this.useMaterial(item);}catch(e){this.toast((e as Error).message);}finally{this.busy=false;}
  }
  private async extract():Promise<void>{if(this.busy)return;const asset=this.assets[this.contentIndex];if(!asset)return;this.busy=true;try{await this.library.cutout(asset.id);this.assets=this.library.items.slice();this.toast('主体已存入素材库');}catch(e){this.toast((e as Error).message);}finally{this.busy=false;}}
  private async tone(index:number):Promise<void>{const photo=this.currentPhoto(),asset=this.library.match(photo?.src??'');if(!photo||!asset||this.busy)return;const id=this.face.id;this.busy=true;
    try{const src=await this.library.tone(asset,index);if(this.screen==='edit'&&this.face.id===id&&this.currentPhoto()?.src===photo.src){this.currentPhoto()!.src=src;this.toneIndex=index;this.publishFace();}}catch(e){this.toast((e as Error).message);}finally{this.busy=false;}}
  private variants():TemplateDefinition[]{return CARD_TEMPLATES.filter(t=>t.kind===templateDefinition(this.face.template?.id??'')?.kind&&t.kind!=='ensemble');}
  private changeDesign(id:string):void{if(!this.face.template||id===this.face.template.id)return;const old=this.face.template;this.face.template=newCardTemplate(id);this.face.template.palette=old.palette;this.face.template.fill=old.fill;
    if(id==='portrait'){if(!this.currentPhoto())this.attachPhoto(this.face,this.assets[0],false);this.face.template.photoMode='photo';this.face.template.photoFraming='cover';}this.publishFace();if(id==='portrait')this.smartClock();}
  private setCap(cap:CapBlock,index:number):void{const old=cardCapabilities(this.face)[index];if(old){replaceCardCapability(this.face,old,cap);this.publishFace();}}
  private setText(value:string):void{const text=this.face.elements.find(e=>e.kind==='text');if(text){text.text=value;this.publishFace();}}
  private setPhotoMode(subject:boolean):void{if(!this.face.template)return;this.face.template.photoMode=subject?'subject':'photo';this.face.template.photoFraming=subject?'fit':'cover';this.face.template.zoom=1;this.face.template.offsetX=0;this.face.template.offsetY=0;this.camera.reset();this.publishFace();}
  private clampPhoto():void{constrainDemoPhoto(this.face);}
  private photoTouch(event:TouchEvent):void{if(this.editTab!==2||this.face.template?.id!=='portrait')return;event.stopPropagation();const t=this.face.template;if(event.type===TouchType.Cancel){this.camera.reset();return;}const excluded=event.type===TouchType.Up?event.changedTouches.map(p=>p.id):[];
    const unit=this.faceWidth()/300,points:PhotoGesturePoint[]=event.touches.filter(p=>!excluded.includes(p.id)).map((p:TouchObject):PhotoGesturePoint=>({id:p.id,x:p.x/unit,y:p.y/unit}));const pose=this.camera.update(points,t,photoControlRegion(this.face));t.zoom=pose.zoom;t.offsetX=pose.offsetX;t.offsetY=pose.offsetY;this.clampPhoto();this.publishFace();}
  private boardWidth():number{return Math.min(this.viewportW-48,420,(this.viewportH-this.safeTop-this.safeBottom-290)*this.document.canvasAspect);}
  private faceWidth():number{return Math.max(120,Math.min(this.viewportW-56,400,(this.viewportH-this.safeTop-this.safeBottom-342)*this.face.w/this.face.h));}
  private thumbWidth():number{return Math.min(174,(this.viewportW-60)/2);}
  private beginMove(id:string):void{this.selected=id;this.checkpoint();this.beforeGesture=JSON.stringify(this.document);}
  private move(id:string,dx:number,dy:number):void{const before=JSON.parse(this.beforeGesture) as FridgeState,origin=before.cards.find(c=>c.id===id);if(!origin)return;
    const sx=344/this.boardWidth(),sy=470/(this.boardWidth()/this.document.canvasAspect);let x=dx*sx,y=dy*sy;
    if(Math.abs(origin.x+x+origin.w/2-172)<6)x=172-origin.w/2-origin.x;
    for(const card of this.document.cards){const start=before.cards.find(c=>c.id===card.id);if(start&&(card.id===id||origin.groupId&&card.groupId===origin.groupId)){card.x=start.x+x;card.y=start.y+y;}}
    this.document=JSON.parse(JSON.stringify(this.document)) as FridgeState;
  }
  private arrangement(action:string):void{const card=this.selectedCard();if(!card)return;this.checkpoint();
    if(action==='front')card.z=++this.document.maxZ;
    if(action==='back')card.z=Math.min(...this.document.cards.map(c=>c.z))-1;
    if(action==='center'){card.x=(344-card.w)/2;card.y=(344/this.document.canvasAspect-card.h)/2/(344/this.document.canvasAspect)*470;}
    if(action==='straight')card.rot=0;
    if(action==='left')card.rot-=5;if(action==='right')card.rot+=5;
    if(action==='small'||action==='large'){const factor=action==='small'?.9:1.1,oldW=card.w,oldH=card.h;card.w=Math.max(60,Math.min(344,card.w*factor));card.h=oldH*card.w/oldW;card.x+=(oldW-card.w)/2;card.y+=(oldH-card.h)/2/(344/this.document.canvasAspect)*470;}
    if(action==='copy'){if(this.document.cards.length>=16){this.toast('每件作品最多 16 张贴纸');return;}const copy=this.store.copyCard(card);copy.x+=12;copy.y+=12;copy.z=++this.document.maxZ;this.document.cards.push(copy);this.selected=copy.id;}
    if(action==='delete'){this.document.cards=this.document.cards.filter(c=>c.id!==card.id);this.selected='';this.sheet=false;}
